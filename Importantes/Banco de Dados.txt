DROP DATABASE IF EXISTS smart_stroller;
CREATE DATABASE smart_stroller;
USE smart_stroller;

/* =========================
   CARRINHOS INTELIGENTES
   ========================= */

CREATE TABLE smart_stroller (
    id_stroller INT PRIMARY KEY AUTO_INCREMENT,
    numero_identificacao VARCHAR(50) UNIQUE NOT NULL,
    nivel_bateria INT,
    status ENUM(
        'Disponivel',
        'Em Uso',
        'Pagamento',
        'Manutencao'
    ) DEFAULT 'Disponivel',
    ultima_sincronizacao DATETIME,
    data_criacao DATETIME,
    data_remocao DATETIME,
    data_atualizacao DATETIME 
);

INSERT INTO smart_stroller (numero_identificacao, nivel_bateria, status, ultima_sincronizacao, data_criacao, data_atualizacao)
VALUES ('STROLLER-01', 100, 'Disponivel', NOW(), NOW(), NOW());

/* =========================
   USUARIOS
   ========================= */
   

CREATE TABLE usuarios (
    id_usuario INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE, -- Adicionado para identificar unicamente o cliente
    cpf VARCHAR(100) UNIQUE,
    telefone VARCHAR(20),
    data_criacao DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    data_remocao DATETIME
);

/* =========================
   CATEGORIAS
   ========================= */
   

CREATE TABLE categorias (
    id_categoria INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(100) NOT NULL UNIQUE,
     data_criacao DATETIME,
    data_remocao DATETIME,
    data_atualizacao DATETIME 
);

INSERT INTO categorias (nome) VALUES

('Limpeza');

/* =========================
   PRODUTOS
   ========================= */
   

CREATE TABLE produtos (
    id_produto INT PRIMARY KEY AUTO_INCREMENT,
    codigo_rfid VARCHAR(100) UNIQUE, -- Perfeito para leitura do Arduino
    codigo_barras VARCHAR(50) UNIQUE,
    nome VARCHAR(150) NOT NULL,
    preco DECIMAL(10,2) NOT NULL,
    estoque INT NOT NULL DEFAULT 0,
    id_categoria INT,
    status ENUM('disponivel', 'em promocao', 'em falta') DEFAULT 'disponivel',
    data_criacao DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    data_remocao DATETIME,

    FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria)
);

/* =========================
   COMPRAS
   ========================= */
 
CREATE TABLE compras (
    id_compra INT PRIMARY KEY AUTO_INCREMENT,
    id_usuario INT NOT NULL,
    id_stroller INT NOT NULL,
    data_inicio DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_fim DATETIME,
    valor_total DECIMAL(10,2) DEFAULT 0.00,
    status ENUM('Em Andamento', 'Pagamento', 'Concluida', 'Cancelada') DEFAULT 'Em Andamento',
    data_criacao DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    data_remocao DATETIME,

    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario),
    FOREIGN KEY (id_stroller) REFERENCES smart_stroller(id_stroller)
);

/* =========================
   ITENS DA COMPRA
   ========================= */
   

CREATE TABLE itens_compra (
    id_item INT PRIMARY KEY AUTO_INCREMENT,
    id_compra INT NOT NULL,
    id_produto INT NOT NULL,
    quantidade INT NOT NULL DEFAULT 1,
    preco_unitario DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    data_criacao DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    data_remocao DATETIME,

    -- Cascade impede o erro 1451 se a compra for deletada
    FOREIGN KEY (id_compra) REFERENCES compras(id_compra) ON DELETE CASCADE, 
    FOREIGN KEY (id_produto) REFERENCES produtos(id_produto)
);

/* =========================
   PAGAMENTOS
   ========================= */
   

CREATE TABLE pagamentos (
    id_pagamento INT PRIMARY KEY AUTO_INCREMENT,
    id_compra INT UNIQUE NOT NULL,

    metodo ENUM(
        'PIX',
        'Credito',
        'Debito',
        'CarteiraDigital'
    ),

    valor DECIMAL(10,2),

    status ENUM(
        'Pendente',
        'Aprovado',
        'Recusado'
    ) DEFAULT 'Pendente',

    data_pagamento DATETIME,
     data_criacao DATETIME,
    data_remocao DATETIME,
    data_atualizacao DATETIME ,

    FOREIGN KEY (id_compra)
        REFERENCES compras(id_compra)
);

/* =========================
   HISTORICO DE USO
   ========================= */
   

CREATE TABLE historico_uso (
    id_uso INT PRIMARY KEY AUTO_INCREMENT,
    id_stroller INT,
    descricao VARCHAR(255),
    data_uso DATETIME DEFAULT CURRENT_TIMESTAMP,
     data_criacao DATETIME,
    data_remocao DATETIME,
    data_atualizacao DATETIME,

    FOREIGN KEY (id_stroller)
        REFERENCES smart_stroller(id_stroller)
);

/* =========================
   LEITURAS RFID
   ========================= */

CREATE TABLE leituras_rfid (
    id_leitura INT PRIMARY KEY AUTO_INCREMENT,

    id_stroller INT NOT NULL,
    id_produto INT NOT NULL,

    data_criacao DATETIME,
    data_remocao DATETIME,
    data_atualizacao DATETIME,

    acao ENUM('Entrada','Saida') NOT NULL,

    data_leitura DATETIME DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (id_stroller)
        REFERENCES smart_stroller(id_stroller),

    FOREIGN KEY (id_produto)
        REFERENCES produtos(id_produto)
);

SET FOREIGN_KEY_CHECKS = 0;
